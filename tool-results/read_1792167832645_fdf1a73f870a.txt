    99→/** عدد → رشته با حداکثر ۲ رقم اعشار (ورودی‌های کاربر) */
   100→function num(v: number): string {
   101→  return Number.isFinite(v) ? String(Math.round(v * 100) / 100) : "—";
   102→}
   103→
   104→/** عدد اختیاری + واحد → سلول متنی؛ null/undefined → «—» */
   105→function fmtNum(
   106→  v: number | null | undefined,
   107→  unit: string,
   108→  digits = 1
   109→): string {
   110→  if (v === null || v === undefined || !Number.isFinite(v)) return "—";
   111→  const s = v.toFixed(digits);
   112→  return unit === "°" || unit === "٪" ? `${s}${unit}` : `${s} ${unit}`;
   113→}

   115→/** تاریخ ISO → رشته شمسی؛ نامعتبر → «—» */
   116→function formatDate(iso: string): string {
   117→  const d = new Date(iso);
   118→  return Number.isNaN(d.getTime()) ? "—" : d.toLocaleDateString("fa-IR");
   119→}

   123→/** ساخت یک عنصر با استایل inline و فرزندان (متن یا نود) */
   124→function el<K extends keyof HTMLElementTagNameMap>(
   125→  tag: K,
   126→  style: StyleMap,
   127→  ...children: Array<Node | string | null | undefined>
   128→): HTMLElementTagNameMap[K] {
   129→  const node = document.createElement(tag);
   130→  Object.assign(node.style, style);
   131→  for (const child of children) {
   132→    if (child === null || child === undefined) continue;
   133→    node.appendChild(
   134→      typeof child === "string" ? document.createTextNode(child) : child
   135→    );
   136→  }
   137→  return node;
   138→}

   140→/** سلول rich → نود (span رنگی/بولد یا text ساده) */
   141→function richNode(cell: Cell): Node {
   142→  if (typeof cell === "string") return document.createTextNode(cell);
   143→  const span = document.createElement("span");
   144→  span.style.color = cell.color ?? "#1c1917";
   145→  if (cell.bold) span.style.fontWeight = "700";
   146→  span.textContent = cell.text;
   147→  return span;
   148→}

   172→/** جدول عمومی با سرستون (اختیاراً ستون‌های وسط‌چین) */
   173→function table(
   174→  headers: string[],
   175→  rows: Cell[][],
   176→  opts?: { centerCols?: number[] }
   177→): HTMLTableElement {
   178→  const t = document.createElement("table");
   179→  Object.assign(t.style, {
   180→    width: "100%",
   181→    borderCollapse: "collapse",
   182→    fontSize: "12.5px",
   183→    marginTop: "2px",
   184→  });
   185→
   186→  const headRow = t.createTHead().insertRow();
   187→  headers.forEach((header, i) => {
   188→    const th = document.createElement("th");
   189→    Object.assign(th.style, {
   190→      border: "1px solid #e4e4e7",
   191→      background: "#f5f5f4",
   192→      color: "#44403c",
   193→      fontWeight: "700",
   194→      fontSize: "12px",
   195→      padding: "7px 12px",
   196→      textAlign: opts?.centerCols?.includes(i) ? "center" : "right",
   197→    });
   198→    th.textContent = header;
   199→    headRow.appendChild(th);
   200→  });
   201→
   202→  const body = t.createTBody();
   203→  for (const row of rows) {
   204→    const tr = body.insertRow();
   205→    row.forEach((cell, i) => {
   206→      const td = tr.insertCell();
   207→      Object.assign(td.style, {
   208→        border: "1px solid #e4e4e7",
   209→        padding: "6px 12px",
   210→        textAlign: opts?.centerCols?.includes(i) ? "center" : "right",
   211→        verticalAlign: "top",
   212→      });
   213→      td.appendChild(richNode(cell));
   214→    });
   215→  }
   216→  return t;
   217→}

   398→      inputs.useJib

   454→/** ج) ردیف‌های «۴. نتایج تحلیل» — مقادیر null → «—» */

   456→  let utilization: Cell = "—";
   457→  const up = result?.utilizationPct;
   458→  if (up !== null && up !== undefined && Number.isFinite(up)) {
   459→    utilization = {
   460→      text: `${up.toFixed(1)}٪`,
   461→      bold: true,
   462→      color: up > 100 ? "#dc2626" : up >= 80 ? "#b45309" : "#16a34a",
   463→    };
   464→  }
   465→
   466→  let hookMargin: Cell = "—";
   467→  const hm = result?.hookHeightMarginM;
   468→  if (hm !== null && hm !== undefined && Number.isFinite(hm)) {
   469→    hookMargin = {
   470→      text: `${hm.toFixed(1)} متر`,
   471→      color: hm < 0 ? "#dc2626" : "#1c1917",
   472→    };
   473→  }

   551→/** خ) بلوک «۷. لودچارت استفاده‌شده» — بج + جدول نزدیک‌ترین سکشن به طول بوم */
   552→function chartBlock(chart: LoadChart | null, boomLengthM: number): HTMLElement {
   553→  const wrap = el("div", {});
   554→  if (!chart || chart.sections.length === 0) {
   555→    wrap.appendChild(h2("۷. لودچارت استفاده‌شده"));
   556→    wrap.appendChild(
   557→      el("div", { fontSize: "12px", color: "#78716c" }, "لودچارتی ثبت نشده")
   558→    );
   559→    return wrap;
   560→  }
   561→
   562→  const isCustom = chart.source === "custom";
   563→  const badge = el(
   564→    "span",
   565→    {
   566→      display: "inline-block",
   567→      padding: "1px 12px",
   568→      borderRadius: "999px",
   569→      fontSize: "11px",
   570→      fontWeight: "700",
   571→      marginRight: "8px",
   572→      verticalAlign: "middle",
   573→      background: isCustom ? "#f59e0b" : "#52525b",
   574→      color: isCustom ? "#1c1917" : "#ffffff",
   575→    },
   576→    isCustom ? "سفارشی" : "مرجع"
   577→  );
   578→
   579→  const title = h2("۷. لودچارت استفاده‌شده");
   580→  title.appendChild(badge);
   581→  wrap.appendChild(title);
   582→
   583→  // نزدیک‌ترین سکشن (طول بوم) به طول بوم لیفت فعلی
   584→  let best = chart.sections[0];
   585→  for (const section of chart.sections) {
   586→    if (
   587→      Math.abs(section.boomLengthM - boomLengthM) <
   588→      Math.abs(best.boomLengthM - boomLengthM)
   589→    ) {
   590→      best = section;
   591→    }
   592→  }
   593→
   594→  const points = [...best.points].sort((a, b) => a.radiusM - b.radiusM);
   595→  const shown = points.slice(0, MAX_CHART_POINTS);
   596→  const rows: Cell[][] = shown.map((p) => [num(p.radiusM), num(p.capacityT)]);
   597→  if (points.length > shown.length) rows.push(["...", "..."]);
   598→
   599→  wrap.appendChild(
   600→    el(
   601→      "div",
   602→      { fontSize: "11.5px", color: "#57534e", marginBottom: "6px" },
   603→      `سکشن انتخاب‌شده: طول بوم ${num(best.boomLengthM)} متر (${points.length} نقطه)`
   604→    )
   605→  );
   606→  wrap.appendChild(
   607→    table(["شعاع (م)", "ظرفیت مجاز (تن)"], rows, { centerCols: [0, 1] })
   608→  );
   609→  return wrap;
   610→}

   754→/**
   755→ * ساخت DOM + رندر کل گزارش به یک Canvas با html2canvas.
   756→ * عنصر موقت همیشه در finally از body حذف می‌شود.
   757→ */
   758→async function renderReportCanvas(data: ReportData): Promise<HTMLCanvasElement> {
   759→  const { default: html2canvas } = await import("html2canvas-pro");
   760→  const element = buildReportElement(data);
   761→  document.body.appendChild(element);
   762→  try {
   763→    await document.fonts.ready;
   764→    return await html2canvas(element, {
   765→      scale: RENDER_SCALE,
   766→      backgroundColor: "#ffffff",
   767→      useCORS: true,
   768→      logging: false,
   769→    });
   770→  } finally {
   771→    element.remove();
   772→  }
   773→}

   775→/**
   776→ * تولید و دانلود گزارش PDF چندصفحه‌ای A4 (فارسی، RTL).
   777→ * فقط در مرورگر صدا زده شود — ارورها با پیام فارسی مجدداً پرتاب می‌شوند.
   778→ */
   779→export async function downloadPDFReport(data: ReportData): Promise<void> {
   780→  try {
   781→    const canvas = await renderReportCanvas(data);
   782→    const { jsPDF } = await import("jspdf");
   783→
   784→    const pdf = new jsPDF({
   785→      orientation: "portrait",
   786→      unit: "mm",
   787→      format: "a4",
   788→      compress: true,
   789→    });
   790→    pdf.setProperties({
   791→      title: `CraneLift Report — ${data.meta.projectId}`,
   792→      creator: "CraneLift Pro 3D",
   793→    });
   794→
   795→    // برش عمودی گزارش به صفحات A4
   796→    const imgWidth = PAGE_W_MM; // میلی‌متر
   797→    const pxPerMm = canvas.width / imgWidth;
   798→    const pageHeightPx = Math.floor(PAGE_H_MM * pxPerMm);
   799→
   800→    let offsetPx = 0;
   801→    let pageIndex = 0;
   802→    while (offsetPx < canvas.height) {
   803→      const sliceHeightPx = Math.min(pageHeightPx, canvas.height - offsetPx);
   804→
   805→      const slice = document.createElement("canvas");
   806→      slice.width = canvas.width;
   807→      slice.height = sliceHeightPx;
   808→      const ctx = slice.getContext("2d");
   809→      if (!ctx) throw new Error("دسترسی به زمینه بوم (2D context) ممکن نیست.");
   810→      ctx.fillStyle = "#ffffff";
   811→      ctx.fillRect(0, 0, slice.width, slice.height);
   812→      ctx.drawImage(canvas, 0, -offsetPx);
   813→
   814→      const sliceHeightMm = sliceHeightPx / pxPerMm;
   815→      if (pageIndex > 0) pdf.addPage();
   816→      pdf.addImage(
   817→        slice.toDataURL("image/jpeg", JPEG_QUALITY),
   818→        "JPEG",
   819→        0,
   820→        0,
   821→        imgWidth,
   822→        sliceHeightMm
   823→      );
   824→
   825→      offsetPx += sliceHeightPx;
   826→      pageIndex += 1;
   827→    }
   828→
   829→    pdf.save(`CraneLift-Report-${data.meta.projectId}.pdf`);
   830→  } catch (err) {
   831→    throw new Error(
   832→      `خطا در ساخت گزارش PDF: ${err instanceof Error ? err.message : String(err)}`
   833→    );
   834→  }
   835→}
