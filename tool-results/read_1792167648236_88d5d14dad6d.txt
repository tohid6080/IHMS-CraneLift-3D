    28→// به‌جای drei Html (که با React 19 خطای unmount می‌دهد) از Sprite با
    29→// CanvasTexture استفاده می‌کنیم — کاملاً داخل WebGL و بدون React root.
    30→function makeTextTexture(
    31→  text: string,
    32→  opts: { color?: string; bg?: string; border?: string }
    33→): { tex: THREE.CanvasTexture; aspect: number } {
    34→  const pad = 16;
    35→  const font = "700 46px Vazirmatn, Tahoma, sans-serif";
    36→  const measure = document.createElement("canvas").getContext("2d");
    37→  measure!.font = font;
    38→  const w = Math.ceil(measure!.measureText(text).width) + pad * 2;
    39→  const h = 72;
    40→  const canvas = document.createElement("canvas");
    41→  canvas.width = w * 2;
    42→  canvas.height = h * 2;
    43→  const ctx = canvas.getContext("2d")!;
    44→  ctx.scale(2, 2);
    45→  // پس‌زمینه گرد
    46→  ctx.fillStyle = opts.bg ?? "rgba(10,10,10,0.78)";
    47→  ctx.beginPath();
    48→  ctx.roundRect(0.5, 0.5, w - 1, h - 1, 10);
    49→  ctx.fill();
    50→  if (opts.border) {
    51→    ctx.strokeStyle = opts.border;
    52→    ctx.lineWidth = 2.5;
    53→    ctx.stroke();
    54→  }
    55→  ctx.font = font;
    56→  ctx.fillStyle = opts.color ?? "#ffffff";
    57→  ctx.textBaseline = "middle";
    58→  ctx.fillText(text, pad, h / 2 + 2);
    59→  const tex = new THREE.CanvasTexture(canvas);
    60→  tex.colorSpace = THREE.SRGBColorSpace;
    61→  tex.anisotropy = 4;
    62→  return { tex, aspect: w / h };
    63→}

    65→function Label3D({
    66→  position,
    67→  text,
    68→  color = "#ffffff",
    69→  bg = "rgba(10,10,10,0.78)",
    70→  border,
    71→  height = 2.1,
    72→}: {
    73→  position: [number, number, number];
    74→  text: string;
    75→  color?: string;
    76→  bg?: string;
    77→  border?: string;
    78→  height?: number;
    79→}) {
    80→  const { tex, aspect } = useMemo(() => makeTextTexture(text, { color, bg, border }), [text, color, bg, border]);
    81→  useEffect(() => () => tex.dispose(), [tex]);
    82→  return (
    83→    <sprite position={position} scale={[height * aspect, height, 1]}>
    84→      <spriteMaterial map={tex} transparent depthTest={false} toneMapped={false} />
    85→    </sprite>
    86→  );
    87→}

    90→function CaptureRegister() {
    91→  const gl = useThree((s) => s.gl);
    92→  const scene = useThree((s) => s.scene);
    93→  const camera = useThree((s) => s.camera);
    94→  useEffect(() => {
    95→    sceneCapture.capture = () => {
    96→      gl.render(scene, camera);
    97→      return gl.domElement.toDataURL("image/png");
    98→    };
    99→    return () => {
   100→      sceneCapture.capture = null;
   101→    };
   102→  }, [gl, scene, camera]);
   103→  return null;
   104→}

   107→function CameraRig({ radius, height }: { radius: number; height: number }) {
   108→  const controls = useThree((s) => s.controls) as unknown as {
   109→    target: THREE.Vector3;
   110→    object: THREE.PerspectiveCamera;
   111→    update: () => void;
   112→  } | null;
   113→  const lastFit = useRef({ radius: 0, height: 0 });
   114→
   115→  useEffect(() => {
   116→    if (!controls) return;
   117→    const first = lastFit.current.radius === 0;
   118→    const dr = Math.abs(radius - lastFit.current.radius);
   119→    const dh = Math.abs(height - lastFit.current.height);
   120→    if (first || dr > 8 || dh > 12) {
   121→      lastFit.current = { radius, height };
   122→      // فاصله دوربین متناسب با بزرگ‌ترین بعد صحنه + مارجین برای قاب کامل
   123→      const dist = Math.max(radius, height * 0.9) * 1.25 + 18;
   124→      controls.target.set(radius * 0.45, Math.max(2, height * 0.32), 0);
   125→      controls.object.position.set(dist * 0.82, Math.max(11, height * 0.6 + 7), dist * 0.82);
   126→      controls.update();
   127→    }
   128→  }, [controls, radius, height]);
   129→  return null;
   130→}

   185→function ForceArrow({
   186→  origin,
   187→  length,
   188→  color,
   189→  dir,
   190→}: {
   191→  origin: [number, number, number];
   192→  length: number;
   193→  color: string;

   462→      <Line points={[[0, H + 3.6, 0], [jibLengthM * 0.45, H + 1.5, 0]]} color="#71717a" lineWidth={1.2} />
   463→      <Line points={[[0, H + 3.6, 0], [-5.6, H + 1.4, 0]]} color="#71717a" lineWidth={1.2} />
   464→      {/* تالی */}
   465→      <mesh position={[radiusM, H + 0.35, 0]} castShadow>
   466→        <boxGeometry args={[0.9, 0.5, 0.9]} />
   467→        <meshStandardMaterial color="#1c1917" metalness={0.5} roughness={0.4} />
   468→      </mesh>
   469→    </group>
   470→  );
   471→}

   473→// ───────────────────────── قلاب، اسلینگ و بار ─────────────────────────
   474→function HookAndLoad({
   475→  inputs,
   476→  hookX,
   477→  hookTopY,
   478→  statusColor,
   479→  showLabels,
   480→  showForces,
   481→  grossT,
   482→}: {
   483→  inputs: LiftInputs;
   484→  hookX: number;
   485→  hookTopY: number;
   486→  statusColor: string;
   487→  showLabels: boolean;
   488→  showForces: boolean;
   489→  grossT: number;
   490→}) {
   491→  const { loadDims: dims, rigging } = inputs;
   492→  const blockY = hookTopY - HOOK_BLOCK_HEIGHT_M;
   493→
   494→  // افت عمودی اسلینگ‌ها (مطابق منطق تحلیل)
   495→  const halfSpan = Math.max(dims.lengthM, dims.widthM) / 2;
   496→  const beta = Math.max(rigging.slingAngleDeg, 10);
   497→  const drop = rigging.useSpreader ? 1.2 + halfSpan * 0.25 : halfSpan * Math.tan(beta * D2R);
   498→
   499→  const loadTopY = Math.max(dims.heightM, blockY - drop - 0.15);
   500→  const loadCenterY = loadTopY - dims.heightM / 2;
   501→  const liftLen = clamp(grossT * 0.12, 1.4, 5);
   502→
   503→  const slingColor = "#e5e5e5";
   504→  const barY = blockY - 1.2;
   505→
   506→  // نقاط اتصال اسلینگ‌ها
   507→  const slingLines: [number, number, number][][] = useMemo(() => {
   508→    if (rigging.useSpreader) {
   509→      const ends = rigging.slingCount === 4 ? 1 : 1;
   510→      const lower: [number, number, number][][] = [];
   511→      // اسلینگ‌های بالایی: قلاب → دو سر تیر
   512→      const up1: [number, number, number][] = [[hookX, blockY, 0], [hookX - halfSpan - 0.3, barY, 0]];
   513→      const up2: [number, number, number][] = [[hookX, blockY, 0], [hookX + halfSpan + 0.3, barY, 0]];
   514→      lower.push(up1, up2);
   515→      if (rigging.slingCount === 4) {
   516→        for (const sx of [-1, 1]) {
   517→          for (const sz of [-1, 1]) {
   518→            lower.push([
   519→              [hookX + sx * (halfSpan + 0.3), barY, 0],
   520→              [hookX + sx * (dims.lengthM / 2), loadTopY, sz * (dims.widthM / 2)],
   521→            ]);
   522→          }
   523→        }
   524→      } else {
   525→        for (const sx of [-1, 1]) {
   526→          lower.push([
   527→            [hookX + sx * (halfSpan + 0.3), barY, 0],
   528→            [hookX + sx * (dims.lengthM / 2), loadTopY, 0],
   529→          ]);
   530→        }
   531→      }
   532→      return lower.slice(0, 2 + ends * rigging.slingCount);
   533→    }
   534→    if (rigging.slingCount === 4) {
   535→      return ([-1, 1] as const).flatMap((sx) =>
   536→        ([-1, 1] as const).map(
   537→          (sz): [number, number, number][] => [
   538→            [hookX, blockY, 0],
   539→            [hookX + sx * (dims.lengthM / 2), loadTopY, sz * (dims.widthM / 2)],
   540→          ]
   541→        )
   542→      );
   543→    }
   544→    return ([-1, 1] as const).map(
   545→      (sx): [number, number, number][] => [
   546→        [hookX, blockY, 0],
   547→        [hookX + sx * (dims.lengthM / 2), loadTopY, 0],
   548→      ]
   549→    );
   550→  }, [rigging.useSpreader, rigging.slingCount, hookX, blockY, barY, halfSpan, loadTopY, dims.lengthM, dims.widthM]);
   551→
   552→  return (
   553→    <group>
   554→      {/* کابل جرثقیل */}
   555→      <Line points={[[hookX, hookTopY, 0], [hookX, blockY + 0.45, 0]]} color="#d4d4d8" lineWidth={1.4} />
   556→      {/* بلاک قلاب */}
   557→      <mesh position={[hookX, blockY, 0]} castShadow>
   558→        <boxGeometry args={[0.55, 0.85, 0.45]} />
   559→        <meshStandardMaterial color="#1c1917" metalness={0.6} roughness={0.35} />
   560→      </mesh>
   561→      <mesh position={[hookX, blockY - 0.55, 0]} castShadow>
   562→        <cylinderGeometry args={[0.07, 0.07, 0.3, 8]} />
   563→        <meshStandardMaterial color="#a1a1aa" metalness={0.8} roughness={0.25} />
   564→      </mesh>
   565→
   566→      {/* تیر پخش‌کننده */}
   567→      {rigging.useSpreader && (
   568→        <mesh position={[hookX, barY, 0]} castShadow>
   569→          <boxGeometry args={[halfSpan * 2 + 0.9, 0.28, 0.28]} />
   570→          <meshStandardMaterial color="#52525b" metalness={0.5} roughness={0.4} />
   571→        </mesh>
   572→      )}
   573→
   574→      {/* اسلینگ‌ها */}
   575→      {slingLines.map((pts, i) => (
   576→        <Line key={i} points={pts} color={slingColor} lineWidth={1.6} />
   577→      ))}
   578→
   579→      {/* بار — رنگ بر اساس وضعیت تحلیل */}
   580→      <mesh position={[hookX, loadCenterY, 0]} castShadow>
   581→        <boxGeometry args={[dims.lengthM, dims.heightM, dims.widthM]} />
   582→        <meshStandardMaterial color={statusColor} metalness={0.25} roughness={0.55} transparent opacity={0.92} />
   583→        <Edges color="#ffffff" />
   584→      </mesh>
   585→
   586→      {/* خطوط نیرو */}
   587→      {showForces && (
   588→        <group>
   589→          {/* وزن بار (به پایین) */}
   590→          <ForceArrow origin={[hookX, loadCenterY + dims.heightM / 2 - 0.1, 0]} length={liftLen} color="#ef4444" dir={-1} />
   591→          {/* نیروی بالابر (به بالا) */}
   592→          <ForceArrow origin={[hookX, loadTopY + 0.1, 0]} length={liftLen * 0.85} color="#22c55e" dir={1} />
   593→        </group>
   594→      )}
   595→
   596→      {/* برچسب‌ها */}
   597→      {showLabels && (
   598→        <>
   599→          <Label3D position={[hookX, loadCenterY, 0]} text={`W = ${grossT.toFixed(1)} t`} />
   600→          <Label3D
   601→            position={[hookX, loadTopY + 0.9, 0]}
   602→            text={`h = ${loadTopY.toFixed(1)} m`}
   603→            color="#111111"
   604→            bg={statusColor}
   605→          />
   606→        </>
   607→      )}
   608→    </group>
   609→  );
   610→}

   612→// ───────────────────────── بدنه اصلی صحنه ─────────────────────────
   613→function SceneContent() {
   614→  const selectedId = useLiftStore((s) => s.selectedCraneId);
   615→  const inputs = useLiftStore((s) => s.inputs);
   616→  const result = useLiftStore((s) => s.result);
   617→  const view = useView3D();
   618→
   619→  const customCranes = useLiftStore((s) => s.customCranes);
   620→  const crane = getCraneById(selectedId, customCranes);
   621→  const geo = useMemo(() => (crane ? solveGeometry(crane, inputs) : null), [crane, inputs]);
   622→
   623→  const statusColor = result ? STATUS_COLORS[result.status] : "#a1a1aa";
   624→  const grossT = inputs.loadWeightT + inputs.hookBlockWeightT + inputs.riggingWeightT;
   625→
   626→  if (!crane || !geo) return null;
   627→
   628→  const isTower = crane.type === "tower";
   629→  const slewRad = -inputs.slewAngleDeg * D2R;
   630→  const slopeRad = inputs.groundSlopeDeg * D2R;
   631→
   632→  // هندسه قلاب (حتی با خطای هندسی، برای رندر fallback داریم)
   633→  const boomLen = inputs.boomLengthM;
   634→  const jibH = inputs.useJib ? inputs.jibLengthM * Math.cos(inputs.jibAngleDeg * D2R) : 0;
   635→  const jibV = inputs.useJib ? inputs.jibLengthM * Math.sin(inputs.jibAngleDeg * D2R) : 0;
   636→  const fbAngle = geo.ok && !isTower ? geo.boomAngleDeg : inputs.boomAngleDeg;
   637→  const hookX = isTower ? geo.radiusM : crane.specs.pivotOffsetM + boomLen * Math.cos(fbAngle * D2R) + jibH;
   638→  const hookY = isTower
   639→    ? (crane.specs.freestandingHookHeightM ?? 50)
   640→    : crane.specs.pivotHeightM + boomLen * Math.sin(fbAngle * D2R) + jibV;
   641→
   642→  const exRadius = Math.max(geo.tipRadiusM || hookX, inputs.boomLengthM) + 3;
   643→
   644→  // مختصات جهانی بار برای خط شعاع
   645→  const loadWorldX = Math.cos(-slewRad) * hookX;
   646→  const loadWorldZ = Math.sin(-slewRad) * hookX;
   647→
   648→  return (
   649→    <>
   650→      <CaptureRegister />
   651→      <CameraRig radius={hookX} height={hookY} />
   652→      <color attach="background" args={["#18181b"]} />
   653→      <fog attach="fog" args={["#18181b", 100, 280]} />
   654→      <hemisphereLight args={["#e7e5e4", "#1c1917", 0.55]} />
   655→      <ambientLight intensity={0.3} />
   656→      <directionalLight
   657→        castShadow
   658→        position={[38, 55, 26]}
   659→        intensity={1.5}
   660→        shadow-mapSize-width={2048}
   661→        shadow-mapSize-height={2048}
   662→        shadow-camera-left={-70}
   663→        shadow-camera-right={70}
   664→        shadow-camera-top={70}
   665→        shadow-camera-bottom={-70}
   666→        shadow-camera-far={220}
   667→      />
   668→      <directionalLight position={[-30, 25, -20]} intensity={0.35} color="#fef3c7" />
   669→
   670→      {/* دنیای شیب‌دار (زمین + جرثقیل + بار) */}
   671→      <group rotation={[0, 0, slopeRad]}>
   672→        <Ground />
   673→        {view.exclusion && <ExclusionZone radiusM={exRadius} />}
   674→
   675→        {/* خط شعاع کاری روی زمین */}
   676→        {view.radiusLine && (
   677→          <group>
   678→            <Line
   679→              points={[[0, 0.06, 0], [loadWorldX, 0.06, loadWorldZ]]}
   680→              color="#f59e0b"
   681→              lineWidth={2}
   682→              dashed
   683→              dashSize={0.9}
   684→              gapSize={0.5}
   685→            />
   686→            {view.labels && (
   687→              <Label3D
   688→                position={[loadWorldX / 2, 0.8, loadWorldZ / 2]}
   689→                text={`R = ${geo.radiusM.toFixed(1)} m`}
   690→                color="#fbbf24"
   691→                bg="rgba(15,10,2,0.78)"
   692→                border="rgba(245,158,11,0.45)"
   693→              />
   694→            )}
   695→          </group>
   696→        )}
   697→
   698→        {/* جرثقیل با دوران سوپراستراکچر */}
   699→        <group rotation={[0, slewRad, 0]}>
   700→          {crane.type === "mobile" && (
   701→            <MobileCrane
   702→              specs={crane.specs}
   703→              boomAngleRad={fbAngle * D2R}
   704→              boomLengthM={boomLen}
   705→              useJib={inputs.useJib}
   706→              jibLengthM={inputs.jibLengthM}
   707→              jibAngleRad={inputs.jibAngleDeg * D2R}
   708→            />
   709→          )}
   710→          {crane.type === "crawler" && (
   711→            <CrawlerCrane
   712→              specs={crane.specs}
   713→              boomAngleRad={fbAngle * D2R}
   714→              boomLengthM={boomLen}
   715→              useJib={inputs.useJib}
   716→              jibLengthM={inputs.jibLengthM}
   717→              jibAngleRad={inputs.jibAngleDeg * D2R}
   718→            />
   719→          )}
   720→          {crane.type === "tower" && (
   721→            <TowerCrane specs={crane.specs} jibLengthM={boomLen} radiusM={geo.radiusM} />
   722→          )}
   723→          <HookAndLoad
   724→            inputs={inputs}
   725→            hookX={hookX}
   726→            hookTopY={hookY}
   727→            statusColor={statusColor}
   728→            showLabels={view.labels}
   729→            showForces={view.forces}
   730→            grossT={grossT}
   731→          />
   732→        </group>
   733→
   734→        {/* مانع */}
   735→        {inputs.obstacle.enabled && (
   736→          <mesh position={[inputs.obstacle.xM, inputs.obstacle.heightM / 2, inputs.obstacle.zM]} castShadow>
   737→            <boxGeometry args={[inputs.obstacle.widthM, inputs.obstacle.heightM, inputs.obstacle.widthM]} />
   738→            <meshStandardMaterial color="#f97316" transparent opacity={0.45} roughness={0.6} />
   739→            <Edges color="#ea580c" />
   740→          </mesh>
   741→        )}
   742→      </group>
   743→
   744→      <OrbitControls
   745→        makeDefault
   746→        enableDamping
   747→        dampingFactor={0.08}
   748→        minDistance={4}
   749→        maxDistance={240}
   750→        maxPolarAngle={Math.PI / 2.04}
   751→      />
   752→    </>
   753→  );
   754→}
