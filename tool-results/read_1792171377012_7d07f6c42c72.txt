    40→function makeTextTexture(
    41→  text: string,
    42→  opts: { color?: string; bg?: string; border?: string }
    43→): { tex: THREE.CanvasTexture; aspect: number } {
    44→  const pad = 16;
    45→  const font = "700 46px Vazirmatn, Tahoma, sans-serif";
    46→  const measure = document.createElement("canvas").getContext("2d");
    47→  measure!.font = font;
    48→  const w = Math.ceil(measure!.measureText(text).width) + pad * 2;
    49→  const h = 72;
    50→  const canvas = document.createElement("canvas");
    51→  canvas.width = w * 2;
    52→  canvas.height = h * 2;
    53→  const ctx = canvas.getContext("2d")!;
    54→  ctx.scale(2, 2);
    55→  // پس‌زمینه گرد
    56→  ctx.fillStyle = opts.bg ?? "rgba(10,10,10,0.78)";
    57→  ctx.beginPath();
    58→  ctx.roundRect(0.5, 0.5, w - 1, h - 1, 10);
    59→  ctx.fill();
    60→  if (opts.border) {
    61→    ctx.strokeStyle = opts.border;
    62→    ctx.lineWidth = 2.5;
    63→    ctx.stroke();
    64→  }
    65→  ctx.font = font;
    66→  ctx.fillStyle = opts.color ?? "#ffffff";
    67→  ctx.textBaseline = "middle";
    68→  ctx.fillText(text, pad, h / 2 + 2);
    69→  const tex = new THREE.CanvasTexture(canvas);
    70→  tex.colorSpace = THREE.SRGBColorSpace;
    71→  tex.anisotropy = 4;
    72→  return { tex, aspect: w / h };
    73→}

   117→function CameraRig({ radius, height, frozen }: { radius: number; height: number; frozen: boolean }) {
   118→  const controls = useThree((s) => s.controls) as unknown as {
   119→    target: THREE.Vector3;
   120→    object: THREE.PerspectiveCamera;
   121→    update: () => void;
   122→  } | null;
   123→  const lastFit = useRef({ radius: 0, height: 0 });
   124→
   125→  useEffect(() => {
   126→    if (!controls || frozen) return;
   127→    const first = lastFit.current.radius === 0;
   128→    const dr = Math.abs(radius - lastFit.current.radius);
   129→    const dh = Math.abs(height - lastFit.current.height);
   130→    if (first || dr > 8 || dh > 12) {
   131→      lastFit.current = { radius, height };
   132→      // فاصله دوربین متناسب با بزرگ‌ترین بعد صحنه + مارجین برای قاب کامل
   133→      const dist = Math.max(radius, height * 0.9) * 1.25 + 18;
   134→      controls.target.set(radius * 0.45, Math.max(2, height * 0.32), 0);
   135→      controls.object.position.set(dist * 0.82, Math.max(11, height * 0.6 + 7), dist * 0.82);
   136→      controls.update();
   137→    }
   138→  }, [controls, radius, height, frozen]);
   139→  return null;
   140→}

   145→function Ground({
   146→  flagMode,
   147→  onGroundClick,
   148→}: {
   149→  flagMode: "off" | "pick" | "place";
   150→  onGroundClick?: (x: number, z: number) => void;
   151→}) {
   152→  const downRef = useRef<{ x: number; y: number } | null>(null);
   153→  return (
   154→    <group>
   155→      <mesh
   156→        rotation={[-Math.PI / 2, 0, 0]}
   157→        position={[0, 0, 0]}
   158→        receiveShadow
   159→        onPointerDown={(e) => {
   160→          downRef.current = { x: e.nativeEvent.clientX, y: e.nativeEvent.clientY };
   161→        }}

   163→          // پرچم فقط با کلیک واقعی (بدون درگ چرخش دوربین) ثبت می‌شود
   164→          if (flagMode === "off" || !onGroundClick) return;

   167→          if (!d) return;
   168→          if (Math.hypot(e.nativeEvent.clientX - d.x, e.nativeEvent.clientY - d.y) > 6) return;
   169→          e.stopPropagation();
   170→          onGroundClick(e.point.x, e.point.z);
   171→        }}
   172→      >
   173→        <planeGeometry args={[260, 260]} />
   174→        <meshStandardMaterial
   175→          color={flagMode === "pick" ? "#33280f" : flagMode === "place" ? "#0f2e22" : "#262626"}
   176→          roughness={1}
   177→          metalness={0}
   178→        />
   179→      </mesh>
   180→      <Grid
   181→        position={[0, 0.02, 0]}
   182→        infiniteGrid
   183→        cellSize={2}
   184→        sectionSize={10}
   185→        cellColor="#3a3a3a"
   186→        sectionColor="#555555"
   187→        fadeDistance={140}

   348→// ───────────────────────── موتور پخش انیمیشن جابجایی ─────────────────────────
   349→function SimDriver() {
   350→  const active = useSim((s) => s.active);
   351→  const speed = useSim((s) => s.speed);
   352→  const lastPush = useRef(0);
   353→  const local = useRef<{ t: number; key: string; tl: ReturnType<typeof buildTimeline> | null }>({
   354→    t: 0,
   355→    key: "",
   356→    tl: null,
   357→  });
   358→
   359→  useFrame((_, dt) => {
   360→    if (!active) return;
   361→    const s = useLiftStore.getState();
   362→    const simS = useSim.getState();
   363→    if (!s.pickPoint || !s.placePoint || !s.transfer) {
   364→      simS.reset();
   365→      return;
   366→    }
   367→    const crane = getCraneById(s.selectedCraneId, s.customCranes);
   368→    if (!crane) {
   369→      simS.reset();
   370→      return;
   371→    }
   372→    const key = `${crane.id}|${s.inputs.boomLengthM}|${s.transfer.checkedAt}|${s.pickPoint.x},${s.pickPoint.z}|${s.placePoint.x},${s.placePoint.z}`;
   373→    if (local.current.key !== key || !local.current.tl) {
   374→      const start = staticPoseOf(crane, s.inputs, s.pickPoint);
   375→      local.current.tl = buildTimeline(crane, s.inputs, s.transfer, start as SegPose);
   376→      local.current.t = 0;
   377→      local.current.key = key;
   378→    }
   379→    local.current.t += Math.min(dt, 0.12) * speed;
   380→    const pose = sampleTimeline(local.current.tl, local.current.t);
   381→    if (!pose) {
   382→      // پایان حرکت — پوز پایانی + نمایش کارت نتیجه
   383→      simS.finish(finalPoseOf(local.current.tl));
   384→      return;
   385→    }
   386→    const now = performance.now();
   387→    if (now - lastPush.current > 33) {
   388→      // ~۳۰Hz — روان و بدون فشار به React
   389→      lastPush.current = now;
   390→      simS.setPose(pose);
   391→    }
   392→  });
   393→  return null;
   394→}

   656→function HookAndLoad({
   657→  inputs,
   658→  hookX,
   659→  hookTopY,
   660→  statusColor,
   661→  showLabels,
   662→  showForces,
   663→  grossT,
   664→  attached,
   665→}: {
   666→  inputs: LiftInputs;
   667→  hookX: number;
   668→  hookTopY: number;
   669→  statusColor: string;
   670→  showLabels: boolean;
   671→  showForces: boolean;
   672→  grossT: number;
   673→  attached: boolean;
   674→}) {
   675→  const { loadDims: dims, rigging } = inputs;
   676→  const blockY = hookTopY - HOOK_BLOCK_HEIGHT_M;
   677→
   678→  // افت عمودی اسلینگ‌ها (مطابق منطق تحلیل)
   679→  const halfSpan = Math.max(dims.lengthM, dims.widthM) / 2;
   680→  const beta = Math.max(rigging.slingAngleDeg, 10);
   681→  const drop = rigging.useSpreader ? 1.2 + halfSpan * 0.25 : halfSpan * Math.tan(beta * D2R);
   682→
   683→  const loadTopY = Math.max(dims.heightM, blockY - drop - 0.15);
   684→  const loadCenterY = loadTopY - dims.heightM / 2;
   685→  const liftLen = clamp(grossT * 0.12, 1.4, 5);
   686→
   687→  const slingColor = "#e5e5e5";
   688→  const barY = blockY - 1.2;
   689→
   690→  // نقاط اتصال اسلینگ‌ها
   691→  const slingLines: [number, number, number][][] = useMemo(() => {
   692→    if (rigging.useSpreader) {
   693→      const ends = rigging.slingCount === 4 ? 1 : 1;
   694→      const lower: [number, number, number][][] = [];
   695→      // اسلینگ‌های بالایی: قلاب → دو سر تیر
   696→      const up1: [number, number, number][] = [[hookX, blockY, 0], [hookX - halfSpan - 0.3, barY, 0]];
   697→      const up2: [number, number, number][] = [[hookX, blockY, 0], [hookX + halfSpan + 0.3, barY, 0]];
   698→      lower.push(up1, up2);
   699→      if (rigging.slingCount === 4) {
   700→        for (const sx of [-1, 1]) {
   701→          for (const sz of [-1, 1]) {
   702→            lower.push([
   703→              [hookX + sx * (halfSpan + 0.3), barY, 0],
   704→              [hookX + sx * (dims.lengthM / 2), loadTopY, sz * (dims.widthM / 2)],
   705→            ]);
   706→          }
   707→        }
   708→      } else {
   709→        for (const sx of [-1, 1]) {
   710→          lower.push([
   711→            [hookX + sx * (halfSpan + 0.3), barY, 0],
   712→            [hookX + sx * (dims.lengthM / 2), loadTopY, 0],
   713→          ]);
   714→        }
   715→      }
   716→      return lower.slice(0, 2 + ends * rigging.slingCount);
   717→    }
   718→    if (rigging.slingCount === 4) {
   719→      return ([-1, 1] as const).flatMap((sx) =>
   720→        ([-1, 1] as const).map(
   721→          (sz): [number, number, number][] => [
   722→            [hookX, blockY, 0],
   723→            [hookX + sx * (dims.lengthM / 2), loadTopY, sz * (dims.widthM / 2)],
   724→          ]
   725→        )
   726→      );
   727→    }
   728→    return ([-1, 1] as const).map(
   729→      (sx): [number, number, number][] => [
   730→        [hookX, blockY, 0],
   731→        [hookX + sx * (dims.lengthM / 2), loadTopY, 0],
   732→      ]
   733→    );
   734→  }, [rigging.useSpreader, rigging.slingCount, hookX, blockY, barY, halfSpan, loadTopY, dims.lengthM, dims.widthM]);
   735→
   736→  return (
   737→    <group>
   738→      {/* کابل جرثقیل */}
   739→      <Line points={[[hookX, hookTopY, 0], [hookX, blockY + 0.45, 0]]} color="#d4d4d8" lineWidth={1.4} />
   740→      {/* بلاک قلاب */}
   741→      <mesh position={[hookX, blockY, 0]} castShadow>
   742→        <boxGeometry args={[0.55, 0.85, 0.45]} />
   743→        <meshStandardMaterial color="#1c1917" metalness={0.6} roughness={0.35} />
   744→      </mesh>
   745→      <mesh position={[hookX, blockY - 0.55, 0]} castShadow>
   746→        <cylinderGeometry args={[0.07, 0.07, 0.3, 8]} />
   747→        <meshStandardMaterial color="#a1a1aa" metalness={0.8} roughness={0.25} />
   748→      </mesh>
   749→
   750→      {/* تیر پخش‌کننده */}
   751→      {rigging.useSpreader && attached && (
   752→        <mesh position={[hookX, barY, 0]} castShadow>
   753→          <boxGeometry args={[halfSpan * 2 + 0.9, 0.28, 0.28]} />
   754→          <meshStandardMaterial color="#52525b" metalness={0.5} roughness={0.4} />
   755→        </mesh>
   756→      )}
   757→
   758→      {/* اسلینگ‌ها — فقط وقتی بار قلاب شده */}
   759→      {attached &&
   760→        slingLines.map((pts, i) => (
   761→          <Line key={i} points={pts} color={slingColor} lineWidth={1.6} />
   762→        ))}
   763→
   764→      {/* بار آویزان — رنگ بر اساس وضعیت تحلیل */}
   765→      {attached && (
   766→        <mesh position={[hookX, loadCenterY, 0]} castShadow>
   767→          <boxGeometry args={[dims.lengthM, dims.heightM, dims.widthM]} />
   768→          <meshStandardMaterial color={statusColor} metalness={0.25} roughness={0.55} transparent opacity={0.92} />
   769→          <Edges color="#ffffff" />
   770→        </mesh>
   771→      )}
   772→
   773→      {/* خطوط نیرو */}
   774→      {showForces && attached && (
   775→        <group>
   776→          {/* وزن بار (به پایین) */}
   777→          <ForceArrow origin={[hookX, loadCenterY + dims.heightM / 2 - 0.1, 0]} length={liftLen} color="#ef4444" dir={-1} />
   778→          {/* نیروی بالابر (به بالا) */}
   779→          <ForceArrow origin={[hookX, loadTopY + 0.1, 0]} length={liftLen * 0.85} color="#22c55e" dir={1} />
   780→        </group>
   781→      )}
   782→
   783→      {/* برچسب‌ها */}
   784→      {showLabels && attached && (
   785→        <>
   786→          <Label3D position={[hookX, loadCenterY, 0]} text={`W = ${grossT.toFixed(1)} t`} />
   787→          <Label3D
   788→            position={[hookX, loadTopY + 0.9, 0]}
   789→            text={`h = ${loadTopY.toFixed(1)} m`}
   790→            color="#111111"
   791→            bg={statusColor}
   792→          />
   793→        </>
   794→      )}
   795→    </group>
   796→  );
   797→}

   800→function SceneContent() {
   801→  const selectedId = useLiftStore((s) => s.selectedCraneId);

   810→  const customCranes = useLiftStore((s) => s.customCranes);

   814→  const effInputs = useMemo(() => {

   818→        radiusMode: "radius" as const,
   819→        radiusM: Math.hypot(pick.x, pick.z),

   824→  }, [pick, inputs, simActive, simPose]);

   827→

   829→  const grossT = inputs.loadWeightT + inputs.hookBlockWeightT + inputs.riggingWeightT;

   846→  const jibV = effInputs.useJib ? effInputs.jibLengthM * Math.sin(effInputs.jibAngleDeg * D2R) : 0;
   847→  const fbAngle = pose ? pose.boomDeg : geo.ok && !isTower ? geo.boomAngleDeg : effInputs.boomAngleDeg;
   848→  const slewRad = pose ? -pose.slewDeg * D2R : -effInputs.slewAngleDeg * D2R;
   849→  const hookX = pose ? pose.hookX : isTower ? geo.radiusM : crane.specs.pivotOffsetM + boomLen * Math.cos(fbAngle * D2R) + jibH;
   850→  const hookY = pose
   851→    ? pose.hookTopY
   852→    : isTower
   853→      ? (crane.specs.freestandingHookHeightM ?? 50)

   900→        shadow-mapSize-width={2048}

   902→        shadow-camera-left={-70}

   905→        shadow-camera-bottom={-70}

   920→
   921→        {/* حلقه شعاع و پرچم نقطه B (جایگذاری) */}
   922→        {place && view.radiusLine && <RadiusRing radiusM={Math.hypot(place.x, place.z)} color="#10b981" />}
   923→        {place && (
   924→          <FlagMarker x={place.x} z={place.z} label="B — جایگذاری" color="#10b981" showLabels={view.labels} />
   925→        )}
   926→
   927→        {/* بار روی زمین (قبل از برداشت یا بعد از جایگذاری) */}
   928→        {loadGround === "A" && pick && (
   929→          <GroundLoad
   930→            x={pick.x}
   931→            z={pick.z}
   932→            groundY={inputs.pickHeightM}
   933→            inputs={inputs}
   934→            color={statusColor}
   935→            showLabels={view.labels}
   936→            grossT={grossT}
   937→          />
   938→        )}
   939→        {loadGround === "B" && place && (
   940→          <GroundLoad
   941→            x={place.x}
   942→            z={place.z}
   943→            groundY={inputs.placeHeightM}
   944→            inputs={inputs}
   945→            color={statusColor}
   946→            showLabels={view.labels}
   947→            grossT={grossT}
   948→          />
   949→        )}
   950→
   951→        {/* خط شعاع کاری روی زمین */}
   952→        {view.radiusLine && (
   953→          <group>
   954→            <Line
   955→              points={[[0, 0.06, 0], [loadWorldX, 0.06, loadWorldZ]]}
   956→              color="#f59e0b"
   957→              lineWidth={2}
   958→              dashed
   959→              dashSize={0.9}
   960→              gapSize={0.5}
   961→            />
   962→            {view.labels && (
   963→              <Label3D
   964→                position={[loadWorldX / 2, 0.8, loadWorldZ / 2]}
   965→                text={`R = ${radiusDisp.toFixed(1)} m`}
   966→                color="#fbbf24"
   967→                bg="rgba(15,10,2,0.78)"
   968→                border="rgba(245,158,11,0.45)"
   969→              />
   970→            )}
   971→          </group>
   972→        )}
   973→
   974→        {/* جرثقیل با دوران سوپراستراکچر */}
   975→        <group rotation={[0, slewRad, 0]}>
   976→          {crane.type === "mobile" && (
   977→            <MobileCrane
   978→              specs={crane.specs}
   979→              boomAngleRad={fbAngle * D2R}
   980→              boomLengthM={boomLen}
   981→              useJib={inputs.useJib}
   982→              jibLengthM={inputs.jibLengthM}
   983→              jibAngleRad={inputs.jibAngleDeg * D2R}
   984→            />
   985→          )}
   986→          {crane.type === "crawler" && (
   987→            <CrawlerCrane
   988→              specs={crane.specs}
   989→              boomAngleRad={fbAngle * D2R}
   990→              boomLengthM={boomLen}
   991→              useJib={inputs.useJib}
   992→              jibLengthM={inputs.jibLengthM}
   993→              jibAngleRad={inputs.jibAngleDeg * D2R}
   994→            />
   995→          )}
   996→          {crane.type === "tower" && (
   997→            <TowerCrane specs={crane.specs} jibLengthM={boomLen} radiusM={hookX} />
   998→          )}
   999→          <HookAndLoad
  1000→            inputs={inputs}
  1001→            hookX={hookX}
  1002→            hookTopY={hookY}
  1003→            statusColor={statusColor}
  1004→            showLabels={view.labels}
  1005→            showForces={view.forces}
  1006→            grossT={grossT}
  1007→            attached={attached}
  1008→          />
  1009→        </group>
  1010→
  1011→        {/* مانع */}
  1012→        {inputs.obstacle.enabled && (
  1013→          <mesh position={[inputs.obstacle.xM, inputs.obstacle.heightM / 2, inputs.obstacle.zM]} castShadow>
  1014→            <boxGeometry args={[inputs.obstacle.widthM, inputs.obstacle.heightM, inputs.obstacle.widthM]} />
  1015→            <meshStandardMaterial color="#f97316" transparent opacity={0.45} roughness={0.6} />
  1016→            <Edges color="#ea580c" />
  1017→          </mesh>
  1018→        )}
  1019→      </group>
  1020→
  1021→      <OrbitControls
  1022→        makeDefault
  1023→        enableDamping
  1024→        dampingFactor={0.08}
  1025→        minDistance={4}
  1026→        maxDistance={240}
  1027→        maxPolarAngle={Math.PI / 2.04}
  1028→      />
  1029→    </>
  1030→  );
  1031→}
